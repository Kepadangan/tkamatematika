    let currentQuestion = 0;
    let userAnswers = new Array(40).fill(null); // Menyimpan indeks jawaban pengguna (0-3)

    // Referensi Elemen DOM
    const introContainer = document.getElementById("intro-container");
    const quizContainer = document.getElementById("quiz-container");
    const resultContainer = document.getElementById("result-container");
    const questionsWrapper = document.getElementById("questions-wrapper");
    const prevBtn = document.getElementById("prev-btn");
    const nextBtn = document.getElementById("next-btn");
    const submitBtn = document.getElementById("submit-btn");
    const progressBar = document.getElementById("progress");

    // Fungsi Memulai Kuis
    function startQuiz() {
        introContainer.style.display = "none";
        quizContainer.style.display = "block";
        buildQuestions();
        showQuestion(0);
    }

    // Fungsi Membuat Elemen Soal ke DOM
    function buildQuestions() {
        questionsWrapper.innerHTML = ""; // Memastikan container kosong sebelum membangun
        questionsData.forEach((q, index) => {
            const card = document.createElement("div");
            card.className = "question-card";
            card.id = `q-${index}`;
            
            const qText = document.createElement("div");
            qText.className = "question-text";
            qText.innerHTML = `${index + 1}. ${q.q}`;
            
            const optionsDiv = document.createElement("div");
            optionsDiv.className = "options";
            
            q.options.forEach((opt, optIndex) => {
                const btn = document.createElement("button");
                btn.className = "option-btn";
                btn.innerHTML = `${String.fromCharCode(65 + optIndex)}. ${opt}`;
                btn.onclick = () => selectOption(index, optIndex, btn);
                optionsDiv.appendChild(btn);
            });

            card.appendChild(qText);
            card.appendChild(optionsDiv);
            questionsWrapper.appendChild(card);
        });
    }

    // Fungsi Menyimpan Jawaban Pilihan
    function selectOption(qIndex, optIndex, btnElement) {
        userAnswers[qIndex] = optIndex;
        
        const currentCard = document.getElementById(`q-${qIndex}`);
        const allBtns = currentCard.querySelectorAll(".option-btn");
        
        allBtns.forEach(btn => btn.classList.remove("selected"));
        btnElement.classList.add("selected");
    }

    // Fungsi Menampilkan Soal Berdasarkan Indeks
    function showQuestion(index) {
        document.querySelectorAll(".question-card").forEach(card => card.classList.remove("active"));
        document.getElementById(`q-${index}`).classList.add("active");
        
        document.getElementById("question-header").innerText = `Soal ${index + 1} dari 40`;
        progressBar.style.width = `${((index + 1) / 40) * 100}%`;

        prevBtn.disabled = index === 0;
        
        if (index === 39) {
            nextBtn.style.display = "none";
            submitBtn.style.display = "inline-block";
        } else {
            nextBtn.style.display = "inline-block";
            submitBtn.style.display = "none";
        }
    }

    // Fungsi berpindah soal
    function changeQuestion(direction) {
        currentQuestion += direction;
        showQuestion(currentQuestion);
    }

    // Fungsi Menghitung Skor dan Menampilkan Ulasan
    function showResult() {
        let score = 0;
        const reviewList = document.getElementById("review-list");
        reviewList.innerHTML = "<h3>Review Jawaban:</h3>";

        questionsData.forEach((q, index) => {
            const isCorrect = userAnswers[index] === q.answer;
            if (isCorrect) score++;

            const revItem = document.createElement("div");
            revItem.className = "review-item";
            
            const status = isCorrect 
                ? `<span class="correct-ans">✔ Benar</span>` 
                : `<span class="wrong-ans">✘ Salah (Kunci: ${String.fromCharCode(65 + q.answer)})</span>`;
            
            const userAnsText = userAnswers[index] !== null 
                ? String.fromCharCode(65 + userAnswers[index]) 
                : "Tidak dijawab";

            revItem.innerHTML = `
                <p><strong>Soal ${index + 1}:</strong> ${q.q}</p>
                <p>Jawaban Anda: ${userAnsText} &mdash; ${status}</p>
            `;
            reviewList.appendChild(revItem);
        });

        const finalScore = Math.round((score / 40) * 100);
        
        quizContainer.style.display = "none";
        resultContainer.style.display = "block";
        
        document.getElementById("final-score").innerText = finalScore;
        document.getElementById("result-summary").innerText = `Anda menjawab benar ${score} dari 40 soal.`;
    }
